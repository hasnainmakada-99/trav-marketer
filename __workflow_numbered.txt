  89: // Levenshtein edit distance between two strings (optimised 1-row version)
  90: function levenshtein(a: string, b: string): number {
  91:   if (a === b) return 0;
  92:   if (a.length === 0) return b.length;
  93:   if (b.length === 0) return a.length;
  94:   const row: number[] = Array.from({ length: b.length + 1 }, (_, i) => i);
  95:   for (let i = 1; i <= a.length; i++) {
  96:     let prev = i;
  97:     for (let j = 1; j <= b.length; j++) {
  98:       const curr =
  99:         a[i - 1] === b[j - 1]
 100:           ? row[j - 1]
 101:           : 1 + Math.min(row[j], prev, row[j - 1]);
 102:       row[j - 1] = prev;
 103:       prev = curr;
 104:     }
 105:     row[b.length] = prev;
 106:   }
 107:   return row[b.length];
 108: }

 110: // Max edit-distance allowed for a keyword based on its length
 111: function maxDist(keyword: string): number {
 112:   const len = keyword.length;
 113:   if (len <= 3) return 0;   // short words: exact
 114:   if (len <= 5) return 1;   // medium: 1 typo
 115:   return 2;                  // long: up to 2 typos
 116: }
 117: 
 118: // Returns true if any whitespace-separated word in `text` is within
 119: // edit-distance of `keyword`, or if `text` contains `keyword` as a substring.
 120: function fuzzyWord(text: string, keyword: string): boolean {
 121:   if (text.includes(keyword)) return true;
 122:   const dist = maxDist(keyword);
 123:   if (dist === 0) return false;
 124:   return text.split(/[\s,]+/).some(
 125:     (w) => Math.abs(w.length - keyword.length) <= dist && levenshtein(w, keyword) <= dist
 126:   );
 127: }
 128: 
 129: // Returns true when text fuzzy-matches any of the supplied keywords
 130: function fuzzyAny(text: string, keywords: string[]): boolean {
 131:   return keywords.some((kw) => fuzzyWord(text, kw));
 132: }

 234: function parseCallbackTime(raw: string): string | null {
 235:   const corrected = autoCorrect(raw);

 237:   const explicit = pick(
 238:     /\b(?:callback|call me|call back|preferred time|available at|prefer|anytime|any time)\s*[:\-]?\s*([a-zA-Z0-9 :/-]{3,40})/i,
 239:     corrected

 243:   // Day + optional time: "Today at 5 PM", "Tomorrow morning", "Saturday afternoon"
 244:   const timeExpr = corrected.match(
 245:     /\b(today|tomorrow|tonight|this\s+(?:morning|afternoon|evening|week|weekend)|monday|tuesday|wednesday|thursday|friday|saturday|sunday)(?:\s+at\s+[\d:]+\s*(?:am|pm)?|\s+(?:morning|afternoon|evening|night|noon))?\b/i
 246:   );

 251:   if (justTime) return justTime[0].trim();
 252: 
 253:   // "anytime", "asap", "as soon as possible", "earliest"
 254:   if (/\b(anytime|any time|asap|earliest|convenient|whenever)\b/i.test(corrected)) {
 255:     return corrected.trim();
 256:   }
 257: 
 258:   return null;

 288: // Try to parse comma-separated inputs:
 289: // Travel: "2 Adults, July, Bangalore, 5 Nights"
 290: // Lead:   "Sini, +91 9876543210, sini@gmail.com"
 291: function tryParseCommaFormat(raw: string, intent: WorkflowIntent, existingSlots: WorkflowSlotMap): WorkflowSlotMap {
 292:   const parts = raw.split(',').map((p) => p.trim()).filter(Boolean);
 293:   if (parts.length < 2) return {};
 294: 
 295:   const slots: WorkflowSlotMap = {};
 296:   const monthRx = /^(january|february|march|april|may|june|july|august|september|october|november|december|jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)\w*(\s+\d{4})?$/i;
 297:   const nightsRx = /^(\d+)\s*nights?$/i;
 298:   const travellersRx = /^(\d+)\s*(adults?|children|kids?|pax|persons?)/i;
 299:   const dateRx = /\d{1,2}(?:st|nd|rd|th)?\s+\w+|\w+\s+\d{1,2}(?:st|nd|rd|th)?/i;
 300:   const hotelRx = /\d\s*star\s*(hotel)?|luxury|premium/i;
 301:   const emailRx = /^[a-z0-9._%+-]+@[a-z0-9.-]+\.[a-z]{2,}$/i;
 302:   const phoneRx = /^\+?\d[\d\s-]{6,14}\d$/;
 303: 
 304:   // Detect parts that contain a month name (in any position, e.g. "june mid", "mid-June")
 305:   const containsMonthRx = /\b(january|february|march|april|may|june|july|august|september|october|november|december|jan|feb|mar|apr|jun|jul|aug|sep|oct|nov|dec)\b/i;
 306: 
 307:   const travellersAccum: string[] = [];
 308:   const unclassified: string[] = [];
 309: 
 310:   for (const part of parts) {
 311:     const lp = normalize(part);
 312:     if (emailRx.test(lp)) {
 313:       if (!slots.email) slots.email = part.trim().toLowerCase();
 314:     } else if (phoneRx.test(part.trim())) {
 315:       if (!slots.phone) slots.phone = part.trim().replace(/\s+/g, '');
 316:     } else if (monthRx.test(lp) || (containsMonthRx.test(part) && /^[a-zA-Z\s-]+$/.test(part))) {
 317:       // Captures "june", "june mid", "mid-June", "early July" etc.
 318:       if (!slots.travel_time) slots.travel_time = part;
 319:     } else if (nightsRx.test(lp)) {
 320:       const m = lp.match(/^(\d+)/);
 321:       if (m) slots.nights = m[1];
 322:     } else if (travellersRx.test(lp)) {
 323:       travellersAccum.push(part);
 324:     } else if (hotelRx.test(lp)) {
 325:       slots.hotel_preference = lp.replace(/\s*hotels?\s*/g, '').trim();
 326:     } else if (!slots.travel_time && dateRx.test(part) && /\d/.test(part)) {
 327:       slots.travel_time = part;
 328:     } else if (/^\d+$/.test(part.trim()) && !travellersAccum.length && !slots.travellers) {
 329:       // Standalone digit (e.g. "2") — treat as traveller count
 330:       travellersAccum.push(part.trim());
 331:     } else {
 332:       unclassified.push(part);
 333:     }
 334:   }
 335: 
 336:   if (travellersAccum.length > 0) {
 337:     slots.travellers = travellersAccum.join(', ');
 338:   }
 339: 
 340:   // If email or phone were extracted, treat a short alpha unclassified part as the name
 341:   if ((slots.email || slots.phone) && !existingSlots.name) {
 342:     const namePart = unclassified.find(
 343:       (p) => /^[a-zA-Z .'-]{2,40}$/.test(p) && !NON_CITY_WORDS.has(normalize(p))
 344:     );
 345:     if (namePart) slots.name = namePart;
 346:   }
 347: 
 348:   // Remaining unclassified strings may be city names — apply blocklist
 349:   const safeCityPart = (candidate: string) => {
 350:     const key = normalize(candidate).trim();
 351:     // Exclude parts that contain a month name — those are travel dates, not cities
 352:     if (containsMonthRx.test(candidate)) return false;
 353:     if (/\d/.test(candidate)) return false;
 354:     return /^[a-zA-Z\s.'-]{2,30}$/.test(candidate) && !NON_CITY_WORDS.has(key);
 355:   };
 356: 
 357:   const cities = unclassified.filter(safeCityPart);
 358:   if (intent === 'plan_holiday') {
 359:     if (existingSlots.destination) {
 360:       // In travel-details step, a single city usually means departure city.
 361:       if (!existingSlots.departure_city && cities[0]) slots.departure_city = cities[0];
 362:     } else if (!existingSlots.destination && cities[0]) {
 363:       slots.destination = cities[0];
 364:       if (!existingSlots.departure_city && cities[1]) slots.departure_city = cities[1];
 365:     }
 366:   }
 367:   if (intent === 'hotels') {
 368:     if (!existingSlots.destination && cities[0]) slots.destination = cities[0];
 369:   }
 370:   if (intent === 'flights') {
 371:     if (!existingSlots.from_city && cities[0]) slots.from_city = cities[0];
 372:     if (!existingSlots.to_city && cities[1]) slots.to_city = cities[1];
 373:   }
 374: 
 375:   return slots;
 376: }

 378: function parseGeneralSlots(
 379:   message: string,
 380:   intent: WorkflowIntent = 'unknown',
 381:   knownSlots: WorkflowSlotMap = {}
 382: ): WorkflowSlotMap {
 383:   const raw = autoCorrect(String(message || '').trim()); // fix typos first
 384:   const text = normalize(raw);
 385:   const slots: WorkflowSlotMap = {};
 386: 
 387:   // New slot types
 388:   const ht = detectHolidayType(raw);
 389:   if (ht) slots.holiday_type = ht;
 390: 
 391:   const hp = parseHotelPreference(raw);
 392:   if (hp) slots.hotel_preference = hp;
 393: 
 394:   const ppa = detectPostPackageAction(raw);
 395:   if (ppa) slots.post_package_action = ppa;
 396: 
 397:   // from → to for flights
 398:   const fromTo = raw.match(/\bfrom\s+([a-zA-Z .'-]+?)\s+to\s+([a-zA-Z .'-]+?)(?:\s|,|$)/i);
 399:   if (fromTo?.[1] && fromTo?.[2]) {
 400:     slots.from_city = fromTo[1].trim();
 401:     slots.to_city = fromTo[2].trim();
 402:   } else {
 403:     // Common compact route input: "Delhi to Mumbai, 25th June, 2 Adults"
 404:     const compactFromTo = raw.match(
 405:       /(?:^|,|\broute\b\s*[:\-]?\s*)([a-zA-Z][a-zA-Z .'-]{1,28})\s+to\s+([a-zA-Z][a-zA-Z .'-]{1,28})(?=,|$|\s+\d)/i
 406:     );
 407:     if (compactFromTo?.[1] && compactFromTo?.[2]) {
 408:       slots.from_city = compactFromTo[1].trim();
 409:       slots.to_city = compactFromTo[2].trim();
 410:     }
 411:   }
 412: 
 413:   if (!slots.destination) {
 414:     // Require colon/dash after "destination" to avoid capturing question phrases
 415:     const destination = pick(
 416:       /\b(?:going to|visit|trip to|holiday to|for hotels in|destination\s*[:\-])\s*([a-zA-Z .'-]{2,40})/i,
 417:       raw
 418:     );
 419:     if (destination) slots.destination = destination;
 420:   }
 421:   if (!slots.destination && slots.to_city) {
 422:     slots.destination = slots.to_city;
 423:   }
 424: 
 425:   // Short standalone response (1-3 alpha words) is likely a destination when in plan_holiday or hotels context
 426:   if ((intent === 'plan_holiday' || intent === 'hotels') && !slots.destination) {
 427:     const words = raw.trim().split(/\s+/);
 428:     const isShortAlpha = words.length <= 3 && words.every((w) => /^[a-zA-Z.'-]+$/.test(w));
 429:     const NOT_DESTINATIONS = new Set([
 430:       'plan a holiday', 'plan holiday', 'flights', 'hotels', 'exclusive', 'personalized',
 431:       'personalised', 'yes', 'no', 'ok', 'okay', 'hi', 'hello', 'hey',
 432:     ]);
 433:     const hasTravelSlots =
 434:       Boolean(slots.travellers) || Boolean(slots.travel_time) ||
 435:       Boolean(slots.departure_city) || Boolean(slots.nights);
 436:     const normalizedRaw = normalize(raw);
 437:     const looksLikeCallback =
 438:       Boolean(slots.post_package_action) ||
 439:       CALLBACK_PHRASES.some((p) => normalizedRaw.includes(p)) ||
 440:       /\b(callback|call me|call back|arrange|support|speak to|contact us|speak to human|help me)\b/i.test(raw);
 441:     if (isShortAlpha && !hasTravelSlots && !looksLikeCallback && !NOT_DESTINATIONS.has(normalizedRaw)) {
 442:       slots.destination = raw.trim();
 443:     }
 444:   }
 445: 
 446:   // Month/date extraction — keyword-prefixed
 447:   const travelTime = pick(
 448:     /\b(?:travel month|travel date|travel dates|date|on|in)\s*[:\-]?\s*([a-zA-Z0-9 ,/-]{3,40})/i,
 449:     raw
 450:   );
 451:   if (travelTime) slots.travel_time = travelTime;
 452: 
 453:   // Month/date extraction — standalone (e.g. "july", "10th July")
 454:   if (!slots.travel_time) {
 455:     const monthMatch = raw.match(
 456:       /\b(january|february|march|april|may|june|july|august|september|october|november|december|jan|feb|mar|apr|jun|jul|aug|sep|oct|nov|dec)(?:\s+\d{4})?\b/i
 457:     );
 458:     if (monthMatch) slots.travel_time = monthMatch[0].trim();
 459:   }
 460:   if (!slots.travel_time) {
 461:     const dateMatch = raw.match(
 462:       /\b(\d{1,2}(?:st|nd|rd|th)?\s+(?:jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)\w*|\b(?:jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)\w*\s+\d{1,2}(?:st|nd|rd|th)?)\b/i
 463:     );
 464:     if (dateMatch) slots.travel_time = dateMatch[0].trim();
 465:   }
 466: 
 467:   // Travellers
 468:   const travFull = raw.match(
 469:     /\b(\d+\s*adults?\s*(?:,?\s*(?:and\s+)?\d+\s*(?:children|kids?|child))?)\b/i
 470:   );
 471:   if (travFull) {
 472:     slots.travellers = travFull[1].trim();
 473:   } else {
 474:     const travellers = pick(
 475:       /\b(?:travellers|travelers|traveller count|pax|passengers)\s*[:\-]?\s*([a-zA-Z0-9 ,/+]{1,30})/i,
 476:       raw
 477:     );
 478:     if (travellers) slots.travellers = travellers;
 479:     else {
 480:       const pax = pick(/\b(\d+)\s*(?:pax|passengers?|adults?)\b/i, raw);
 481:       if (pax) slots.travellers = pax;
 482:     }
 483:   }
 484: 
 485:   const departure = pick(
 486:     /\b(?:departure city|departing from|from city)\s*[:\-]?\s*([a-zA-Z .'-]{2,40})/i,
 487:     raw
 488:   );
 489:   if (departure) slots.departure_city = departure;
 490:   if (!slots.departure_city && slots.from_city) {
 491:     slots.departure_city = slots.from_city;
 492:   }
 493:   // "from [city]" pattern — catches "from Ahmedabad", "from Delhi" in holiday/hotel context
 494:   if (!slots.departure_city && (intent === 'plan_holiday' || intent === 'hotels')) {
 495:     const fromM = raw.match(/\bfrom\s+([a-zA-Z][a-zA-Z .'-]{1,28})(?=[\s,]|$)/i);
 496:     if (fromM) {
 497:       const candidate = fromM[1].trim();
 498:       const key = normalize(candidate);
 499:       const hasMonth = /\b(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec|january|february|march|april|june|july|august|september|october|november|december)\b/i.test(candidate);
 500:       if (!hasMonth && !NON_CITY_WORDS.has(key) && !/\d/.test(candidate)) {
 501:         slots.departure_city = candidate;
 502:       }
 503:     }
 504:   }
 505: 
 506:   const nights = pick(/\b(\d+)\s*(?:nights?|night stay)\b/i, raw);
 507:   if (nights) slots.nights = nights;
 508: 
 509:   const checkIn = pick(/\b(?:check[\s-]?in)\s*[:\-]?\s*([a-zA-Z0-9 ,/-]{3,30})/i, raw);
 510:   const checkOut = pick(/\b(?:check[\s-]?out)\s*[:\-]?\s*([a-zA-Z0-9 ,/-]{3,30})/i, raw);
 511:   if (checkIn) slots.checkin = checkIn;
 512:   if (checkOut) slots.checkout = checkOut;
 513: 
 514:   const budget = pick(
 515:     /\b(?:budget|approx budget)\s*[:\-]?\s*(?:₹|inr)?\s*([0-9., ]+\s*[kKmM]?)\b/i,
 516:     raw
 517:   );
 518:   if (budget) slots.budget_inr = budget.toUpperCase().replace(/\s+/g, ' ');
 519: 
 520:   const email = pick(/\b([a-z0-9._%+-]+@[a-z0-9.-]+\.[a-z]{2,})\b/i, text);
 521:   if (email) slots.email = email;
 522: 
 523:   const phone = pick(/\b(\+?\d[\d\s-]{7,14}\d)\b/, raw);
 524:   if (phone) slots.phone = phone.replace(/\s+/g, '');
 525: 
 526:   if (!slots.name) {
 527:     const name = pick(/\b(?:name|full name|i am|i'm|my name is)\s*[:\-]?\s*([a-zA-Z .'-]{2,40})/i, raw);
 528:     if (name) slots.name = name;
 529:   }
 530:   // Name from comma format is handled in tryParseCommaFormat
 531: 
 532:   const callbackTime = parseCallbackTime(raw);
 533:   if (callbackTime) slots.callback_time = callbackTime;
 534: 
 535:   // Try comma-separated format if we still have missing common fields
 536:   const commaSlots = tryParseCommaFormat(raw, intent, { ...knownSlots, ...slots });
 537:   return mergeSlots(slots, commaSlots);
 538: }

 540: function mergeSlots(base: WorkflowSlotMap, next: WorkflowSlotMap): WorkflowSlotMap {
 541:   const merged = { ...base };
 542:   for (const [key, val] of Object.entries(next)) {
 543:     if (val && !merged[key as keyof WorkflowSlotMap]) {
 544:       (merged as Record<string, string>)[key] = val;
 545:     }
 546:   }
 547:   return merged;
 548: }

 552: function resolveStage(intent: WorkflowIntent, slots: WorkflowSlotMap): WorkflowStage {
 553:   if (intent === 'unknown') return 'unknown';
 554: 
 555:   // Standalone callback request (no prior travel service)
 556:   if (intent === 'callback_request') {
 557:     if (!slots.name || !slots.phone) return 'collect_lead';
 558:     return slots.callback_time ? 'confirmed' : 'ask_callback';
 559:   }
 560: 
 561:   // If user explicitly requests a callback at ANY stage, skip travel collection
 562:   // and go straight to collecting contact info → callback time.
 563:   if (slots.post_package_action === 'arrange_callback') {
 564:     const hasRequiredLead = Boolean(slots.name && slots.phone);
 565:     if (!hasRequiredLead) return 'collect_lead';
 566:     return slots.callback_time ? 'confirmed' : 'ask_callback';
 567:   }
 568: 
 569:   if (intent === 'plan_holiday') {
 570:     if (!slots.destination) return 'ask_destination';
 571:     if (!slots.holiday_type) return 'ask_holiday_type';
 572: 
 573:     const travelMissing =
 574:       !slots.travellers || !slots.travel_time || !slots.departure_city || !slots.nights;
 575:     const needsHotelPref = slots.holiday_type === 'personalized' && !slots.hotel_preference;
 576: 
 577:     if (travelMissing || needsHotelPref) return 'ask_travel_details';
 578: 
 579:     const hasPostAction = Boolean(slots.post_package_action);
 580:     const hasContactStart = Boolean(slots.name || slots.phone);
 581:     const hasRequiredLead = Boolean(slots.name && slots.phone);
 582: 
 583:     if (hasPostAction || hasContactStart) {
 584:       if (!hasRequiredLead) return 'collect_lead';
 585:       return slots.callback_time ? 'confirmed' : 'ask_callback';
 586:     }
 587: 
 588:     return 'show_packages';
 589:   }
 590: 
 591:   if (intent === 'flights') {
 592:     if (!slots.from_city || !slots.to_city || !slots.travel_time || !slots.travellers) {
 593:       return 'ask_travel_details';
 594:     }
 595:     const hasPostAction = Boolean(slots.post_package_action);
 596:     const hasContactStart = Boolean(slots.name || slots.phone);
 597:     const hasRequiredLead = Boolean(slots.name && slots.phone);
 598:     if (hasPostAction || hasContactStart) {
 599:       if (!hasRequiredLead) return 'collect_lead';
 600:       return slots.callback_time ? 'confirmed' : 'ask_callback';
 601:     }
 602:     return 'show_packages';
 603:   }
 604: 
 605:   if (intent === 'hotels') {
 606:     if (!slots.destination || !slots.travellers || !slots.travel_time || !slots.nights) {
 607:       return 'ask_travel_details';
 608:     }
 609:     const hasPostAction = Boolean(slots.post_package_action);
 610:     const hasContactStart = Boolean(slots.name || slots.phone);
 611:     const hasRequiredLead = Boolean(slots.name && slots.phone);
 612:     if (hasPostAction || hasContactStart) {
 613:       if (!hasRequiredLead) return 'collect_lead';
 614:       return slots.callback_time ? 'confirmed' : 'ask_callback';
 615:     }
 616:     return 'show_packages';
 617:   }
 618: 
 619:   // Other intents: transfer, forex, visa, insurance, mice, booking_status
 620:   if (!slots.name || !slots.phone) return 'collect_lead';
 621:   return slots.callback_time ? 'confirmed' : 'ask_callback';
 622: }

 635: // Returns true when the user's message looks like a question or off-topic text
 636: // that the AI should handle with a brief answer before re-asking the required info.
 637: export function isQuestionLike(message: string): boolean {
 638:   const t = normalize(message);
 639:   if (t.endsWith('?')) return true;
 640:   if (/^(what|how|why|when|where|who|which|can|is|are|do|does|will|should|could|would|tell me)\b/.test(t)) return true;
 641:   if (/\b(tell me|explain|information about|info about|what about)\b/.test(t)) return true;
 642:   return false;
 643: }

 645: function findLockedIntentFromHistory(historyMessages: string[]): WorkflowIntent | null {
 646:   for (let i = historyMessages.length - 1; i >= 0; i--) {
 647:     const msg = historyMessages[i] || '';
 648:     // A greeting marks the start of a new session — stop scanning here so old
 649:     // session messages cannot bleed their intent into the current session.
 650:     if (isGreetingLike(msg)) break;
 651:     const intent = detectWorkflowIntent(msg);
 652:     if (intent !== 'unknown') return intent;
 653:   }
 654:   return null;
 655: }

 657: // Returns the index in historyMessages from which to START parsing slots.
 658: // Scopes slot accumulation to the current session + current service selection
 659: // so old-session data and switched-service data never pollute the current flow.
 660: function findSlotStartIndex(historyMessages: string[], intent: WorkflowIntent): number {
 661:   // Step 1: last greeting = session boundary
 662:   let sessionStart = 0;
 663:   for (let i = historyMessages.length - 1; i >= 0; i--) {
 664:     if (isGreetingLike(historyMessages[i])) {
 665:       sessionStart = i + 1;
 666:       break;
 667:     }
 668:   }
 669: 
 670:   // Step 2: within session, find last explicit service selection for this intent.
 671:   // E.g. if user did Hotels → then switched to "Plan a Holiday", we start
 672:   // accumulating slots from the "Plan a Holiday" message, ignoring the Hotels part.
 673:   for (let i = historyMessages.length - 1; i >= sessionStart; i--) {
 674:     const msg = historyMessages[i];
 675:     if (isDirectServiceSelection(msg) && detectWorkflowIntent(msg) === intent) {
 676:       return i; // include this message (defines intent) onward
 677:     }
 678:   }
 679: 
 680:   return sessionStart;
 681: }
 682: 
 683: // Returns true only when the message is an explicit, unambiguous service selection
 684: // (button tap, single keyword, or "Plan a Holiday" / "Flights" / "Hotels").
 685: // Used to decide whether the current message can override a locked intent from history.
 686: function isDirectServiceSelection(message: string): boolean {
 687:   const text = normalizeSelectionText(message);
 688:   if (/^(svc_1|svc_2|svc_3|1|2|3)$/.test(text)) return true;
 689:   if (/^(plan a holiday|plan holiday|flights?|hotels?|visa|transfer|forex|insurance|mice|booking status)$/.test(text)) {
 690:     return true;
 691:   }
 692:   // Accept common interactive payload variants like:
 693:   // "1 plan a holiday", "2 flights", "3 hotels"
 694:   if (/^(1|2|3)\s+(plan a holiday|plan holiday|flights?|hotels?)$/.test(text)) return true;
 695:   return false;
 696: }
 697: 
 698: // Primary menu/service taps should start a fresh requirement-collection flow.
 699: // (Digits intentionally excluded so "1/2/3" can still act as package selection in show_packages.)
 700: function isPrimaryServiceSelection(message: string): boolean {
 701:   const text = normalizeSelectionText(message);
 702:   if (/^(svc_1|svc_2|svc_3|plan a holiday|plan holiday|flights?|hotels?)$/.test(text)) return true;
 703:   // Handle interactive labels that include numeric prefixes.
 704:   if (/^(1|2|3)\s+(plan a holiday|plan holiday|flights?|hotels?)$/.test(text)) return true;
 705:   return false;
 706: }

 708: export function resolveWorkflowState(args: {
 709:   userMessage: string;
 710:   classifiedIntent?: string;
 711:   selectedIntent?: WorkflowIntent | null;
 712:   aiSlots?: WorkflowSlotMap;
 713:   historyMessages?: string[];
 714: }): WorkflowState {
 715:   const historyMessages = args.historyMessages || [];
 716:   const msgIntent = detectWorkflowIntent(args.userMessage, args.classifiedIntent);
 717:   const selected = args.selectedIntent && args.selectedIntent !== 'unknown' ? args.selectedIntent : null;
 718:   const locked = (shouldResetState(args.userMessage) || isGreetingLike(args.userMessage))
 719:     ? null
 720:     : findLockedIntentFromHistory(historyMessages);
 721: 
 722:   // Locked intent from history wins unless the current message is an explicit,
 723:   // unambiguous service selection (button tap / single keyword).
 724:   // This prevents travel-detail messages like "2 Adults, July, Bangalore, 5 Nights"
 725:   // from accidentally overriding the locked plan_holiday intent.
 726:   const canOverrideLocked = !locked || isDirectServiceSelection(args.userMessage);
 727:   // resolvedOverride: prefer explicitly-passed selectedIntent, fall back to detected msgIntent.
 728:   // This ensures that when the user taps "Hotels" and locked='plan_holiday', the Hotels
 729:   // intent wins (selected is null but msgIntent='hotels', so resolvedOverride='hotels').
 730:   const resolvedOverride: WorkflowIntent | null =
 731:     selected ?? (msgIntent !== 'unknown' ? msgIntent : null);
 732:   const intent: WorkflowIntent =
 733:     (canOverrideLocked && resolvedOverride)
 734:       ? resolvedOverride
 735:       : locked
 736:         ? locked
 737:         : selected || (msgIntent !== 'unknown' ? msgIntent : 'unknown');
 738:   const source: WorkflowState['source'] = (canOverrideLocked && resolvedOverride)
 739:     ? (selected ? 'selected_now' : 'detected_now')
 740:     : locked
 741:       ? 'locked_history'
 742:       : 'detected_now';
 743: 
 744:   if (intent === 'unknown') {
 745:     return { intent, stage: 'unknown', source, slots: {}, missingSlots: [], complete: false, leadShouldBeSaved: false };
 746:   }
 747: 
 748:   // Only accumulate slots from the current session + current service selection.
 749:   // Prevents old-session data (e.g. previous travellers/dates/post_package_action)
 750:   // from polluting a fresh conversation and making the bot skip stages.
 751:   const slotStart = isPrimaryServiceSelection(args.userMessage)
 752:     ? historyMessages.length
 753:     : findSlotStartIndex(historyMessages, intent);
 754:   const sessionMessages = historyMessages.slice(slotStart);
 755: 
 756:   // Lead slots (name, phone, email, callback_time) require special scoping.
 757:   // They must NOT be pulled from session history when the user has just triggered
 758:   // lead collection (by tapping Arrange Callback / Get Details) — old-session contact
 759:   // info would otherwise skip the collect_lead stage entirely.
 760:   // Strategy:
 761:   // This is a sample change at line 741 to make sure oracle is keeping up with the new changes
 762:   //   1. Parse only travel/service slots from full session history.
 763:   //   2. Find the last message in session history that set post_package_action.
 764:   //   3. Only if that anchor exists (we're CONTINUING lead collection, not starting it),
 765:   //      pull lead slots from messages strictly AFTER the anchor — and only if no
 766:   //      service-selection reset happened after it.
 767:   //   4. Always parse all slots from the current message.
 768:   const LEAD_SLOT_KEYS = ['name', 'phone', 'email', 'callback_time'] as const;
 769: 
 770:   let slots: WorkflowSlotMap = {};
 771:   for (const item of sessionMessages) {
 772:     const parsed = parseGeneralSlots(item, intent, slots);
 773:     const travelOnly = { ...parsed };
 774:     for (const k of LEAD_SLOT_KEYS) delete travelOnly[k];
 775:     slots = mergeSlots(slots, travelOnly);
 776:   }
 777: 
 778:   // Find last post_package_action anchor in session history
 779:   let lastPpaIdx = -1;
 780:   for (let i = sessionMessages.length - 1; i >= 0; i--) {
 781:     const p = parseGeneralSlots(sessionMessages[i], intent, {});
 782:     if (p.post_package_action) { lastPpaIdx = i; break; }
 783:   }
 784:   if (lastPpaIdx !== -1) {
 785:     // Verify no service-selection reset happened after the anchor
 786:     const afterAnchor = sessionMessages.slice(lastPpaIdx + 1);
 787:     const resetAfter = afterAnchor.some(m => isDirectServiceSelection(m) || isGreetingLike(m));
 788:     if (!resetAfter) {
 789:       for (const item of afterAnchor) {
 790:         const parsed = parseGeneralSlots(item, intent, slots);
 791:         for (const k of LEAD_SLOT_KEYS) {
 792:           if (parsed[k] && !slots[k]) (slots as Record<string, string>)[k] = parsed[k] as string;
 793:         }
 794:       }
 795:     }
 796:   }
 797: 
 798:   slots = mergeSlots(slots, parseGeneralSlots(args.userMessage, intent, slots));
 799:   // AI-extracted slot hints are a fallback layer: they only fill still-missing fields
 800:   // and never override slots already captured by deterministic parsing.
 801:   if (args.aiSlots) {
 802:     slots = mergeSlots(slots, args.aiSlots);
 803:   }
 804: 
 805:   // SAFETY: Prevent lead bleed from skipping collect_lead.
 806:   // If the current message sets post_package_action AND no lead has been
 807:   // explicitly provided AFTER the last post_package_action in session history,
 808:   // clear any accidentally accumulated lead slots so the bot always asks fresh.
 809:   const currentMsgPpa = parseGeneralSlots(args.userMessage, intent, {}).post_package_action;
 810:   if (currentMsgPpa) {
 811:     const msgsAfterLastPpa = lastPpaIdx !== -1
 812:       ? sessionMessages.slice(lastPpaIdx + 1)
 813:       : [];
 814:     const leadProvidedAfterPpa = msgsAfterLastPpa.some(m => {
 815:       const p = parseGeneralSlots(m, intent, {});
 816:       return Boolean(p.name || p.phone || p.email);
 817:     });
 818:     if (!leadProvidedAfterPpa) {
 819:       delete slots.name;
 820:       delete slots.phone;
 821:       delete slots.email;
 822:       delete slots.callback_time;
 823:     }
 824:   }
 825: 
 826:   // If "customize" is selected as post_package_action, switch to personalized
 827:   if (slots.post_package_action === 'customize') {
 828:     slots.holiday_type = 'personalized';
 829:   }
 830: 
 831:   // Stage-aware overrides — applied BEFORE final stage resolution.
 832:   const preStage = resolveStage(intent, slots);
 833: 
 834:   // show_packages: detect bare number/keyword selections as post_package_action
 835:   // so the flow advances to collect_lead instead of repeating the packages.
 836:   if (preStage === 'show_packages' && !slots.post_package_action) {
 837:     const msg = args.userMessage.trim();
 838:     const msgL = normalize(msg);
 839:     if (/^[123]$/.test(msg) ||
 840:         /\b(budget|premium|luxury|package [123]|option [123]|deal [123])\b/i.test(msg) ||
 841:         /\b(get package|get detail|package detail)\b/i.test(msg) ||
 842:         /\b(itinerary|day wise|day-wise|day by day)\b/i.test(msg)) {
 843:       slots.post_package_action = 'get_details';
 844:     } else if (/\b(customis|customiz|modify|edit plan|change plan)\b/i.test(msgL)) {
 845:       slots.post_package_action = 'customize';
 846:     } else if (/\b(callback|call me|call back|arrange call)\b/i.test(msgL)) {
 847:       slots.post_package_action = 'arrange_callback';
 848:     }
 849:   }
 850: 
 851:   // Callback fallback: treat the entire short message as callback time when
 852:   // all lead info is collected but callback_time is still missing.
 853:   // Handles freeform responses like "evening tomorrow" or "5 pm on saturday".
 854:   if (preStage === 'ask_callback' && !slots.callback_time) {
 855:     const msg = args.userMessage.trim();
 856:     const msgL = normalize(msg);
 857:     if (msg && msg.length <= 60) {
 858:       const isCallbackActionOnly =
 859:         slots.post_package_action === 'arrange_callback' &&
 860:         /\b(arrange\s*callback|callback|call\s*back|call me)\b/i.test(msgL);
 861:       // Only use whole message if it doesn't look like a new service request
 862:       const looksLikeNewIntent =
 863:         detectWorkflowIntent(msg) !== 'unknown' || isGreetingLike(msg);
 864:       if (!looksLikeNewIntent && !isCallbackActionOnly) {
 865:         slots.callback_time = msg;
 866:       }
 867:     }
 868:   }
 869: 
 870:   const stage = resolveStage(intent, slots);
 871:   const complete = stage === 'confirmed';
 872:   const leadShouldBeSaved = complete;
 873: 
 874:   // missingSlots kept for backward compat — list what's still needed
 875:   const missingSlots: string[] = [];
 876:   if (intent === 'plan_holiday') {
 877:     if (!slots.destination) missingSlots.push('destination');
 878:     if (!slots.holiday_type) missingSlots.push('holiday_type');
 879:     if (!slots.travellers) missingSlots.push('travellers');
 880:     if (!slots.travel_time) missingSlots.push('travel_time');
 881:     if (!slots.departure_city) missingSlots.push('departure_city');
 882:     if (!slots.nights) missingSlots.push('nights');
 883:   } else if (intent === 'flights') {
 884:     if (!slots.from_city) missingSlots.push('from_city');
 885:     if (!slots.to_city) missingSlots.push('to_city');
 886:     if (!slots.travel_time) missingSlots.push('travel_time');
 887:     if (!slots.travellers) missingSlots.push('travellers');
 888:   } else if (intent === 'hotels') {
 889:     if (!slots.destination) missingSlots.push('destination');
 890:     if (!slots.travellers) missingSlots.push('travellers');
 891:     if (!slots.travel_time) missingSlots.push('travel_time');
 892:     if (!slots.nights) missingSlots.push('nights');
 893:   } else if (intent === 'callback_request') {
 894:     // only lead slots matter
 895:   }
 896:   if (!slots.name) missingSlots.push('name');
 897:   if (!slots.phone) missingSlots.push('phone');
 898:   if (!slots.callback_time) missingSlots.push('callback_time');
 899: 
 900:   return { intent, stage, source, slots, missingSlots, complete, leadShouldBeSaved };
 901: }

1042: export function detectWorkflowIntent(message: string, classifiedIntent?: string): WorkflowIntent {
1043:   const raw = autoCorrect(message);           // fix typos before detection
1044:   const text = normalize(raw);
1045:   const intentText = normalize(classifiedIntent || '');
1046:   const joined = `${text} ${intentText}`;
1047: 
1048:   // Digit menu selections (1/2/3) only fire when the ENTIRE message is that digit.
1049:   // This prevents "2 adults" or "2, june, ..." from being classified as "Flights".
1050:   const isDigitSelect = (n: string) => new RegExp(`^${n}\\.?\\s*$`).test(text);
1051: 
1052:   const hasFlightKeyword =
1053:     /(^|\s)(svc_2|flight|flights|fligth|fligths|air ticket|airticket|airfare)(\s|$)/.test(joined);
1054: 
1055:   // plan_holiday — fuzzy-match "holiday", "package", "tour", "leisure"
1056:   if (
1057:     isDigitSelect('1') ||
1058:     /(^|\s)(svc_1|plan holiday|plan a holiday|holiday package|holiday|leisure|tour package)(\s|$)/.test(joined) ||
1059:     fuzzyAny(text, ['holiday', 'package', 'tour', 'leisure'])
1060:   ) {
1061:     // Explicit digit-1, svc_1, or the phrase "plan a holiday" always wins — even if the
1062:     // message also contains "flights"/"hotels" (e.g. the user echoing the quick-reply menu).
1063:     const explicitPlanHoliday =
1064:       isDigitSelect('1') ||
1065:       /(^|\s)(svc_1|plan a holiday|plan holiday)(\s|$)/.test(text);
1066:     const flightLike = hasFlightKeyword;
1067:     const hotelLike = isDigitSelect('3') || /(^|\s)(svc_3|hotel|hotels|stay|resort)(\s|$)/.test(joined);
1068:     if (explicitPlanHoliday || (!flightLike && !hotelLike)) return 'plan_holiday';
1069:   }
1070: 
1071:   // flights — strict keyword matching only (avoid false positives like "nights")
1072:   if (
1073:     isDigitSelect('2') ||
1074:     hasFlightKeyword
1075:   ) {
1076:     return 'flights';
1077:   }
1078: 
1079:   // hotels
1080:   if (isDigitSelect('3') || /(^|\s)(svc_3|hotel|hotels|stay|resort)(\s|$)/.test(joined)) {
1081:     return 'hotels';
1082:   }
1083: 
1084:   // Multi-slot travel data → plan_holiday
1085:   if (
1086:     /(destination|travel month|travel date|travel dates|budget|package)/.test(joined) &&
1087:     /(travellers|travelers|adults?|pax|nights?)/.test(joined)
1088:   ) {
1089:     return 'plan_holiday';
1090:   }
1091:   if (/\bfrom\b.+\bto\b.+(\bpassengers?\b|\bpax\b|\badults?\b)/.test(joined)) {
1092:     return 'flights';
1093:   }
1094: 
1095:   if (/(airport transfer|transfer|pickup|drop)/.test(joined)) return 'transfer';
1096:   if (fuzzyAny(text, ['forex']) || /(currency exchange|money exchange)/.test(joined)) return 'forex';
1097:   if (fuzzyAny(text, ['visa'])) return 'visa';
1098:   if (fuzzyAny(text, ['insurance'])) return 'insurance';
1099:   if (/(mice|corporate event|meetings incentives conferences exhibitions)/.test(joined)) return 'mice';
1100:   if (/(booking status|status|pnr|reference|booking id)/.test(joined)) return 'booking_status';
1101: 
1102:   // ── NEW: standalone callback request ──
1103:   // Match explicit phrases like "arrange callback", "call me", "callback please" etc.
1104:   // Only if no other service intent has been triggered.
1105:   if (
1106:     /\b(arrange\s*(a\s)?callback|call\s*me|call\s*back|callback\s*request|request\s*callback|get\s*callback)\b/i.test(joined) ||
1107:     (/\bcallback\b/i.test(joined) && joined.length < 30)  // short message containing only "callback"
1108:   ) {
1109:     return 'callback_request';
1110:   }
1111: 
1112:   return 'unknown';
1113: }
