 124: function verifyYCloudSignature(rawBody: string, signatureHeader: string, secret: string): boolean {
 125:   const parts = signatureHeader
 126:     .split(',')
 127:     .map((part) => part.trim())
 128:     .filter(Boolean);
 129:   const timestamp = parts.find((part) => part.startsWith('t='))?.slice(2);
 130:   const signatureHex = parts.find((part) => part.startsWith('s='))?.slice(2);
 131:   if (!timestamp || !signatureHex) return false;
 132: 
 133:   const signedPayload = `${timestamp}.${rawBody}`;
 134:   const expectedHex = createHmac('sha256', secret).update(signedPayload).digest('hex');
 135:   const expected = Buffer.from(expectedHex, 'hex');
 136:   const provided = Buffer.from(signatureHex, 'hex');
 137:   if (expected.length !== provided.length) return false;
 138:   return timingSafeEqual(expected, provided);
 139: }

 148: async function withTimeout<T>(promise: Promise<T>, timeoutMs: number, fallback: T): Promise<T> {
 149:   let timeoutHandle: ReturnType<typeof setTimeout> | null = null;
 150:   try {
 151:     return await Promise.race([
 152:       promise,
 153:       new Promise<T>((resolve) => {
 154:         timeoutHandle = setTimeout(() => resolve(fallback), Math.max(300, timeoutMs));
 155:       }),
 156:     ]);
 157:   } finally {
 158:     if (timeoutHandle) clearTimeout(timeoutHandle);
 159:   }
 160: }

 198: async function hasHumanTakeover(teamId: string, phone: string) {
 199:   if (DISABLE_HUMAN_HANDOVER) {
 200:     return false;

 202:   const rows = await listDocuments('conversations', [
 203:     Query.equal('teamId', teamId),
 204:     Query.equal('phone', phone),
 205:     Query.orderDesc('$createdAt'),
 206:     Query.limit(100),
 207:   ]);
 208: 
 209:   let latestStaffTs = 0;
 210:   let latestAiTs = 0;
 211: 
 212:   for (const doc of rows.documents as Array<{
 213:     sentBy?: string;
 214:     createdAt?: string;
 215:     $createdAt?: string;
 216:   }>) {
 217:     const ts = new Date(doc.createdAt || doc.$createdAt || 0).getTime();
 218:     if (!Number.isFinite(ts) || ts <= 0) continue;
 219: 
 220:     if (doc.sentBy === 'staff' && ts > latestStaffTs) {
 221:       latestStaffTs = ts;
 222:     }
 223:     if (doc.sentBy === 'ai' && ts > latestAiTs) {
 224:       latestAiTs = ts;
 225:     }
 226:   }
 227: 
 228:   if (!(latestStaffTs > 0 && latestStaffTs > latestAiTs)) {
 229:     return false;
 230:   }
 231: 
 232:   const safeMinutes =
 233:     Number.isFinite(HUMAN_HANDOVER_MINUTES) && HUMAN_HANDOVER_MINUTES > 0
 234:       ? HUMAN_HANDOVER_MINUTES
 235:       : 45;
 236:   const handoverWindowMs = safeMinutes * 60 * 1000;
 237:   return Date.now() - latestStaffTs <= handoverWindowMs;

 532:  */
 533: async function processIncomingMessage(
 534:   message: Parameters<typeof extractMessage>[0],
 535:   webhookPhoneNumberId: string,
 536:   requestUrl: string
 537: ) {
 538:   try {
 539:     const incoming = extractMessage(message);
 540:     if (!incoming || !incoming.phone) {
 541:       console.warn('[WhatsApp] Skipping message: missing sender phone');
 542:       return;
 543:     }
 544: 
 545:     const phone = incoming.phone;
 546:     const type = incoming.type || 'text';
 547:     const text = 'text' in incoming ? (incoming.text || '') : '';
 548:     const messageId = incoming.messageId || null;
 549: 
 550:     // Drop messages from the Traventions own number (self-loop) and auto-replies
 551:     const ownPhone = (process.env.YCLOUD_WHATSAPP_FROM || '').replace(/\D/g, '');
 552:     if (ownPhone && phone.replace(/\D/g, '') === ownPhone) {
 553:       console.log(`[WhatsApp] Skipping self-message from own number ${phone}`);
 554:       return;
 555:     }
 556:     const AUTOREPLY_PATTERNS = [
 557:       /reach out to us/i, /thanks for reaching out/i, /out of (office|town)/i,
 558:       /automated (message|reply|response)/i, /do not (reply|respond)/i,
 559:       /this is an? (auto|automated)/i, /unsubscribe/i,
 560:     ];
 561:     if (text && AUTOREPLY_PATTERNS.some(p => p.test(text))) {
 562:       console.log(`[WhatsApp] Skipping auto-reply from ${phone}: "${text.slice(0, 80)}"`);
 563:       return;
 564:     }
 565: 
 566:     const teamId = await resolveTeamIdByPhoneNumberId(webhookPhoneNumberId);
 567:     const dedupeKey = buildInboundDedupeKey({
 568:       teamId,
 569:       phone,
 570:       messageId,
 571:       type,
 572:       text,
 573:     });
 574: 
 575:     if (wasRecentlyProcessedInbound(dedupeKey)) {
 576:       console.log(`[WhatsApp] Skipping duplicate inbound (memory cache) ${dedupeKey}`);
 577:       return;
 578:     }
 579: 
 580:     let existingInbound: { customerId?: string; createdAt?: string; $createdAt?: string } | null = null;
 581:     if (messageId) {
 582:       const existing = await listDocuments('conversations', [
 583:         Query.equal('teamId', teamId),
 584:         Query.equal('phone', phone),
 585:         Query.equal('sentBy', 'customer'),
 586:         Query.equal('metaMessageId', messageId),
 587:         Query.limit(1),
 588:       ]);
 589:       if (existing.documents.length > 0) {
 590:         existingInbound = existing.documents[0] as {
 591:           customerId?: string;
 592:           createdAt?: string;
 593:           $createdAt?: string;
 594:         };
 595:         const inboundTs = new Date(
 596:           existingInbound.createdAt || existingInbound.$createdAt || 0
 597:         ).getTime();
